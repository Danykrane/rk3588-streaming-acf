cmake_minimum_required(VERSION 3.23)
project(StreamingACF VERSION 1.0.0 LANGUAGES CXX)

include(CTest)
option(ACF_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/ext/range-v3/CMakeLists.txt")
    message(FATAL_ERROR "Run git submodule update --init --recursive to fetch range-v3")
endif()
set(RANGES_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(RANGES_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(RANGES_BUILD_PERF OFF CACHE BOOL "" FORCE)
add_subdirectory(ext/range-v3 EXCLUDE_FROM_ALL)
# Keep strict warnings for our code, not for third-party headers.
target_include_directories(range-v3 SYSTEM INTERFACE "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/ext/range-v3/include>")

add_library(acf src/Autocorrelation.cpp)
add_library(acf::acf ALIAS acf)
target_include_directories(acf PUBLIC include)
target_compile_features(acf PUBLIC cxx_std_20)
target_link_libraries(acf PRIVATE range-v3::range-v3)

add_executable(acf_cli src/main.cpp)
target_link_libraries(acf_cli PRIVATE acf::acf)

function(acf_checks target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    if(ACF_SANITIZERS)
        if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
            message(FATAL_ERROR "ACF_SANITIZERS requires Clang or GCC")
        endif()
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        # Static-library consumers also need the sanitizer runtime at link time.
        target_link_options(${target} PUBLIC -fsanitize=address,undefined)
    endif()
endfunction()
acf_checks(acf)
acf_checks(acf_cli)

if(BUILD_TESTING)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/ext/doctest/CMakeLists.txt")
        message(FATAL_ERROR "Run git submodule update --init --recursive to fetch doctest")
    endif()
    set(DOCTEST_WITH_TESTS OFF CACHE BOOL "" FORCE)
    add_subdirectory(ext/doctest EXCLUDE_FROM_ALL)
    add_executable(acf_tests tests/AutocorrelationTests.cpp)
    target_link_libraries(acf_tests PRIVATE acf::acf doctest::doctest)
    target_include_directories(acf_tests SYSTEM PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/ext/doctest")
    acf_checks(acf_tests)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang" AND CMAKE_CXX_COMPILER_VERSION VERSION_GREATER_EQUAL 22)
        # doctest 2.4.12 generates unique identifiers with __COUNTER__.
        target_compile_options(acf_tests PRIVATE -Wno-c2y-extensions)
    endif()
    include(ext/doctest/scripts/cmake/doctest.cmake)
    doctest_discover_tests(acf_tests)
    add_test(NAME cli_input COMMAND ${CMAKE_COMMAND} -DCLI=$<TARGET_FILE:acf_cli> -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/Cli.cmake)
    add_test(NAME cli_help COMMAND acf_cli --help)
    add_test(NAME cli_unknown_option COMMAND acf_cli --unknown)
    set_tests_properties(cli_unknown_option PROPERTIES WILL_FAIL TRUE)
endif()
